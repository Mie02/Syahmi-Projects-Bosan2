This code runs natively on Android (API 26–34), intercepting incoming WhatsApp/Telegram notifications and firing automated replies via RemoteInput without root access:

// AutoLeadNotificationListener.kt package com.syahmi.autolead.service import android.service.notification.NotificationListenerService import android.service.notification.StatusBarNotification import android.app.RemoteInput import android.content.Intent import android.os.Bundle class AutoLeadNotificationListener : NotificationListenerService() { override fun onNotificationPosted(sbn: StatusBarNotification) { val packageName = sbn.packageName if (packageName != "com.whatsapp" && packageName != "com.whatsapp.w4b") return val extras = sbn.notification.extras val senderTitle = extras.getString("android.title") ?: return val messageText = extras.getCharSequence("android.text")?.toString() ?: return // Prevent echo loops if (senderTitle == "You" || senderTitle == "Anda") return // Execute Rule Matching Engine val matchedReply = AutomationRuleEngine.matchRule(messageText) ?: return // Dispatch Quick Reply via WearableExtender / RemoteInput sbn.notification.actions?.forEach { action -> action.remoteInputs?.forEach { remoteInput -> val intent = Intent() val bundle = Bundle() bundle.putCharSequence(remoteInput.resultKey, matchedReply.text) RemoteInput.addResultsToIntent(arrayOf(remoteInput), intent, bundle) action.actionIntent.send(this, 0, intent) // Log lead into local Room Database & queue Google Sheets Sync LeadRepository.recordLead(senderTitle, messageText, matchedReply.text) } } } }
